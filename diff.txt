// Utility: get selected word inside input
function getSelectedWord() {
  const input = promptInput;
  const start = input.selectionStart;
  const end = input.selectionEnd;
  if (start === end) return null;
  const text = input.value;
  return text.substring(start, end);
// Utility: replace selected word with synonym
function replaceSelectedWord(newWord) {
  const input = promptInput;
  const start = input.selectionStart;
  const end = input.selectionEnd;
  const before = input.value.substring(0, start);
  const after = input.value.substring(end);
  input.value = before + newWord + after;
  // Move cursor to end of inserted word
  const cursorPos = before.length + newWord.length;
  input.setSelectionRange(cursorPos, cursorPos);
  input.focus();
