impl<T: Config> Pallet<T> {
    fn run_offchain_worker(_block_number: BlockNumberFor<T>) -> Result<(), &'static str> {
        // 1. Fetch all active Vault-lane leases from storage
        let leases: Vec<Lease<T>> = Leases::<T>::iter()
            .filter(|(_, lease)| lease.role == ROLE_VAULT_ADAPTER)
            .map(|(_, lease)| lease)
            .collect();

        // 2. Compute 8NN for each addr5d
        for lease in leases.iter() {
            let addr = lease.addr5d.clone();
            let neighbors = Self::compute_8nn(&addr, &leases);

            // 3. Build route hints for each neighbor
            let hints = neighbors
                .into_iter()
                .map(|n| RouteHint::<T> {
                    from_addr5d: addr.clone(),
                    to_addr5d: n.addr5d.clone(),
                    hops: vec![n.addr5d.clone()], // start simple: direct hop
                })
                .collect::<Vec<_>>();

            // 4. Submit unsigned tx to update RouteHints on-chain
            Self::submit_route_hints_update(addr.clone(), hints)?;
        }

        Ok(())
    }

    fn compute_8nn(
        addr: &BoundedVec<u8, T::MaxAddr5dLength>,
        leases: &[Lease<T>],
    ) -> Vec<Lease<T>> {
        let mut scored = leases
            .iter()
            .filter(|l| &l.addr5d != addr)
            .map(|l| {
                let d = Self::distance(addr, &l.addr5d);
                (d, l.clone())
            })
            .collect::<Vec<_>>();

        scored.sort_by_key(|(d, _)| *d);
        scored.into_iter().take(8).map(|(_, l)| l).collect()
    }

    fn distance(
        a: &BoundedVec<u8, T::MaxAddr5dLength>,
        b: &BoundedVec<u8, T::MaxAddr5dLength>,
    ) -> u32 {
        // simple Hamming distance over bytes
        a.iter()
            .zip(b.iter())
            .map(|(x, y)| if x == y { 0 } else { 1 })
            .sum()
    }

    fn submit_route_hints_update(
        addr: BoundedVec<u8, T::MaxAddr5dLength>,
        hints: Vec<RouteHint<T>>,
    ) -> Result<(), &'static str> {
        use frame_system::offchain::SubmitTransaction;
        let call = Call::<T>::push_route_hint_batch { addr5d: addr, hints };
        SubmitTransaction::<T, Call<T>>::submit_unsigned_transaction(call.into())
            .map_err(|_| "submit failed")
    }
}

fn encode_shortcode(addr5d: &str) -> String {
    let digest = hash_addr5d(addr5d);
    // treat digest as big integer, emit base32
    let mut value = 0u128;
    for b in digest.iter() {
        value = (value << 8) | (*b as u128);
    }

    let mut chars = Vec::new();
    for _ in 0..5 {
        let idx = (value & 0x1F) as usize; // 5 bits
        chars.push(ALPHABET[idx]);
        value >>= 5;
    }

    chars.reverse(); // most significant first
    String::from_utf8(chars).unwrap()
}
